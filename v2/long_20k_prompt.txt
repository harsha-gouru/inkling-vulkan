struct WriteAheadLogConfig_000 {
    uint64_t shard_id = 0;
    uint32_t max_bytes = 4096;
    uint32_t retry_limit = 2;
    bool sync_on_flush = true;
};
Status WriteAheadLog_000::append_record(Slice key, Slice val, uint64_t seq) {
    if (key.size() == 0 || val.size() > cfg_.max_bytes) return Status::InvalidArgument("bounds");
    uint32_t tag = checksum_crc32c(key.data(), key.size(), seq ^ 0x00000000u);
    entries_.push_back({seq, tag, key.size(), val.size()});
    bytes_written_ += key.size() + val.size() + 16;
    return Status::OK();
}
